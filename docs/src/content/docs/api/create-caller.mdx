export const metadata = { title: "createCaller", description: "Factory function that creates a typed caller for server-side direct HTTP calls — no proxy hop needed.", mode: "direct" }

## Import

```ts
import { createCaller } from "@noverax/courier"
```

## Signature

`createCaller` takes your API definition and optional defaults, returning a `TypedClient`. Direct mode calls the backend directly without going through a proxy.

```ts
const caller = createCaller(api, defaults?)
```

## Parameters

### api

An API source definition created with `defineApi()`.

```ts
import { api } from "@/api"

const caller = createCaller(api)
```

### CallerDefaults

Optional default settings applied to every request.

```ts
type CallerDefaults = {
  timeout?: number
  headers?: Record<string, string>
  context?: Context
  onError?: (error: HTTPError) => void | Promise<void>
  onSuccess?: (data: unknown) => void | Promise<void>
  ws?: CallerWSOptions
}

type CallerWSOptions = {
  getAuthToken?: () => string | Promise<string>
  backoff?: { initialMs?: number; maxMs?: number; factor?: number; jitter?: boolean }
  heartbeat?: { intervalMs?: number; timeoutMs?: number }
  messageQueue?: { maxCount?: number; maxBytes?: number }
  protocols?: string[]
  maxMessageSize?: number
}
```

- `ws` — options of the sockets this caller's channels and push subscriptions open; see [Connection options](/docs/advanced/realtime#connection-options).

## Return Type

A call resolves to the response data. Through `.raw` it resolves to the whole response, `HttpResponse<R>`, with `status`, `headers` and `data`; with `.with({ stream: true })` the data is the unread `ReadableStream`.

```ts
type HttpResponse<T> = {
  status: number
  headers: Headers
  data: T
}
```

## Basic Usage

```ts title="lib/caller.ts"
import { createCaller } from "@noverax/courier"
import { api } from "@/api"

export const caller = createCaller(api, {
  timeout: 10000,
})

// Usage: caller.group.method(variables?, context?)
const result = await caller.users.getAll()
const user = await caller.users.getById({ id: "123" })
```

## Passing Context

The optional third argument is a `Context` object. Use it to forward headers from the incoming request to the backend: `authorization`, `user-agent`, `referer`, `origin` and `x-request-id`, the headers the source and the endpoint name, and the user's cookies where they name `cookie`:

```ts
type Context = {
  req?: Request
  res?: Response
  headers?: Headers
}
```

```ts title="app/users/page.tsx"
import { headers } from "next/headers"

// Pass context for cookie/header forwarding
const result = await caller.users.getProfile(
  undefined,
  { headers: await headers() },
)

// With variables AND context
const user = await caller.users.getById(
  { id: "123" },
  { headers: await headers() },
)
```

## Server Components

Direct mode is designed for server-only execution. Use it in React Server Components, API routes, server actions, or any backend runtime.

```ts title="app/users/page.tsx"
import { caller } from "@/lib/caller"

export default async function UsersPage() {
  const result = await caller.users.getAll()

  return (
    <ul>
      {result.data?.map((user) => (
        <li key={user.id}>{user.name}</li>
      ))}
    </ul>
  )
}
```

## Server Actions

```ts title="app/users/actions.ts"
"use server"

import { caller } from "@/lib/caller"
import { revalidatePath } from "next/cache"

export async function createUser(formData: FormData) {
  const name = formData.get("name") as string
  await caller.users.create({ name })
  revalidatePath("/users")
}
```

## Configuring Defaults

```ts title="lib/caller.ts"
const caller = createCaller(api, {
  timeout: 15000,
  headers: {
    "X-Internal-Service": "courier-app",
  },
  onError: (error) => {
    console.error(`[Direct] HTTP ${error.status}:`, error.data)
  },
})
```
