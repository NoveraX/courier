## Overview

This page lists all TypeScript types used across the Nexus library. Types are organized by category — from the high-level request functions down to the internal method types.

## Request Function Types

### RequestFn

The curried request function type returned by `createClientRequest`, `createDirectRequest`, and `createQueryRequest`.

```ts
type RequestFn<T extends Record<string, any>, D extends boolean = false> = <
  K extends keyof T,
  RD extends boolean = false,
>(
  source: K,
  options?: CallOptions<RD>,
) => <
  Q extends EndpointKey<T[K]>,
  M extends ResolveEndpoint<T[K], Q>,
  R = MethodQueryResponse<M extends Endpoint<any, any> ? M : unknown>,
>(
  ...arguments_: RequestArguments<Q, M, D>
) => MethodRequestResponse<RD, R>

// D = false → Proxy mode → returns Promise<R>
// D = true  → Direct mode → returns Promise<Partial<HttpResponse<R>>>
```

### ClientRequestDefaults

Options for `createClientRequest` and `createQueryRequest`.

```ts
type ClientRequestDefaults = {
  routePath?: string
  timeout?: number
  credentials?: RequestCredentials
  headers?: Record<string, string>
  onError?: (error: HTTPError) => void | Promise<void>
  onSuccess?: (data: unknown) => void | Promise<void>
}
```

### DirectRequestDefaults

Options for `createDirectRequest`.

```ts
type DirectRequestDefaults = {
  timeout?: number
  headers?: Record<string, string>
  onError?: (error: HTTPError) => void | Promise<void>

```

## Method Definition Types

### Method

The type of a query method definition. Generic over response type `R` and variables type `V`.

```ts
type Endpoint<R, V = undefined> = (
  methods: Methods,
  variables: V,
) => HttpResponse<R> | Promise<HttpResponse<R>>

// R = response type
// V = variables type (undefined if no variables needed)

// Usage in source definition:
type UserMethods = {
  getAll: Endpoint<User[]>                    // no variables
  getById: Endpoint<User, { id: string }>     // requires { id: string }
  create: Endpoint<User, CreateUserInput>     // requires CreateUserInput
}
```

### Methods

```ts
// HTTP method names — matches ky methods, with "del" instead of "delete"
type Methods = "get" | "post" | "put" | "patch" | "del" | "head"
```

### MethodsInstance

The object your method definitions receive. Contains all six HTTP method functions.

```ts
type MethodsInstance = {
  [K in Methods]: <T = unknown>(
    ...arguments_: MethodParameters<K>
  ) => Promise<HttpResponse<T>>
}

// The object your method definitions receive as the first argument
```

## Response Types

### HttpResponse

```ts
type HttpResponse<T = unknown> = {
  status: number    // HTTP status code (e.g., 200, 404)
  headers: Headers  // Response headers
  data: T           // Parsed response body
}
```

### SuccessResponse

Convenience type for standard success response wrappers.

```ts
type SuccessResponse<D = unknown> = {
  success: true
  data: D
}
```

## Options Types

### CallOptions

Per-request options passed when calling a source method.

```ts
type CallOptions<R extends boolean = false> = Omit<
  SimpleOptions,
  "onUploadProgress"
> & {
  rawData?: R           // Return { status, headers, data } instead of just data
  passVariables?: boolean    // Pass variables to methodsInstance (default: true)
  stream?: boolean      // Return ReadableStream response
  onUploadProgress?: (progress: UploadProgress) => void  // XHR upload progress
}
```

### MethodParameters

```ts
// Parameters for HTTP methods
type MethodParameters<T extends Methods> = T extends "head"
  ? [url?: string, options?: MethodOptions<T>]
  : [url?: string, variables?: Record<string, unknown>, options?: MethodOptions<T>]

// Head has no body/variables, other methods accept variables
```

### MethodOptions

```ts
// Per-call options (subset of ky Options)
type MethodOptions<T extends Methods> = T extends "delete" | "get" | "head"
  ? SimpleOptions
  : OptionsWithSearchParams

type SimpleOptions = Omit<OptionsWithSearchParams, "body" | "json" | "searchParams">

type OptionsWithSearchParams = Omit<Options, "searchParams"> & {
  searchParams?: Record<string, unknown>
}
```

### GenerateMethodOptions

```ts
type GenerateMethodOptions = SimpleOptions & {
  vars?: FormData | Record<string, unknown>
  stream?: boolean
}
```

## Source & Config Types

### BaseOptions

The core configuration type for `Request`.

```ts
type BaseOptions = MergeOptions &
  Omit<Options, "body" | "json" | "prefixUrl" | "searchParams"> & {
    baseURL?: string
    responseType?: ResponseType
    responseParser?: (response: Response) => Promise<unknown>
    defaultData?: Record<string, unknown>
    defaultQueryParams?: Record<string, unknown>
    allowedHeaders?: string[]
    bodySerializer?: (data: Record<string, unknown>) => BodyInit
    searchParamsSerializer?: SearchParamsSerializer
    deduplicate?: boolean
    onRequest?: (info: HookInfo) => HookInfo | undefined | Promise<HookInfo | undefined>
    onResponse?: (
      info: HookInfo,
      response: { status: number; data: unknown },
    ) => { status: number; data: unknown } | undefined | Promise<...>
    onError?: (info: HookInfo, error: HTTPError) => undefined | Promise<undefined>
  }
```

## Query Utility Types

Internal utility types that power the type inference engine. These resolve source names, group names, and method signatures into fully typed request calls.

```ts
// Utility types for query/method resolution

// Extracts all valid query strings for a source
type EndpointKey<T> = ClassMethods<T> | TransformToGroup<Groups<T>>
// Result: "methodName" | "group:methodName"

// Resolves the actual Endpoint type from a query string
type ResolveEndpoint<T, Q> = /* resolves to Endpoint<R, V> */

// Extracts response type from an Endpoint
type MethodQueryResponse<M extends Endpoint<any, any>> = M extends Endpoint<infer R, any> ? R : unknown

// Extracts variables type from an Endpoint
type MethodQueryVariables<M extends Endpoint<any, any>> = M extends Endpoint<any, infer V> ? V : unknown

// Determines arguments based on mode and variables
type RequestArguments<Q, M, D> =
  D extends false
    ? V extends undefined ? [Q] : [Q, V]                  // Proxy mode
    : V extends undefined ? [Q, Context?] : [Q, V, Context?]  // Direct mode

// Return type based on rawData flag
type MethodRequestResponse<RD, R> = Promise<
  RD extends true ? Partial<HttpResponse<R>> : R
>
```

<Callout variant="info" title="Type inference">
These utility types are what make Nexus fully type-safe. When you call `request("api")("users:getById", { id })`, TypeScript resolves the source, group, method, variables, and response types automatically.
</Callout>

## React Query Types

### NexusQueryOptions & NexusMutationOptions

```ts
// React Query integration types

type NexusQueryOptions<R, V> = WithVariables<V> &
  Omit<UseQueryOptions<R, Error, R, QueryKey>, "queryKey" | "queryFn"> & {
    requestOptions?: CallOptions
  }

type NexusMutationOptions<R, V> = Omit<
  UseMutationOptions<R, Error, [V] extends [undefined] ? void : V>,
  "mutationFn"
> & {
  requestOptions?: CallOptions
}

type WithVariables<V> = [V] extends [undefined]
  ? { variables?: undefined }
  : { variables: V }
```
