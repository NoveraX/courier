export const metadata = { title: "RequestGroup", description: "Class that groups related methods under a namespace with a shared URL prefix and optional dynamic path parameters." }

## Import

```ts
import { EndpointGroup } from "@noverax/nexus/datasource"
```

## Signature

```ts
class EndpointGroup {
  constructor(public options?: EndpointGroupOptions)
}
```

## RequestGroupOptions

```ts
type EndpointGroupOptions = MergeOptions & {
  prefix?: string
}

type MergeOptions = {
  pathOptions?: PathOptions
}

type PathOptions = (
  request?: NextRequest,
) => PathOptionsResult | Promise<PathOptionsResult>

type PathOptionsResult =
  | Record<string, boolean | number | string | null | undefined>
  | undefined
```

- `prefix` — URL path prefix prepended to all method URLs in this group. Supports `:param` placeholders resolved by `pathOptions`.
- `pathOptions` — async function that returns dynamic URL parameters. Receives the `NextRequest` object in proxy mode. Values are substituted into `:param` placeholders in the prefix and method URLs.

## Basic Usage

```ts title="api/sources.ts"
import { Datasource, EndpointGroup } from "@noverax/nexus/datasource"

const api = new Datasource({
  baseURL: process.env.API_URL,
  groups: {
    users: new EndpointGroup({ prefix: "/users" }),
    posts: new EndpointGroup({ prefix: "/posts" }),
    auth: new EndpointGroup({ prefix: "/auth" }),
  },
})

// Calling "users:getAll" → GET {baseURL}/users
// Calling "posts:getById" → GET {baseURL}/posts/:id
```

## Prefix

The prefix is prepended to all method URLs within the group. It forms the base path for the group's endpoints.

```ts
// The prefix is prepended to all method URLs within this group
const usersGroup = new EndpointGroup({ prefix: "/api/v2/users" })

// Method URL: "/:id"      → /api/v2/users/:id
// Method URL: "/search"   → /api/v2/users/search
// Method URL: ""          → /api/v2/users
```

## Dynamic Path Options

`pathOptions` is a function that returns key-value pairs for URL parameter substitution. Any `:param` in the prefix or method URL is replaced with the corresponding value.

```ts title="api/sources.ts"
import { EndpointGroup } from "@noverax/nexus/datasource"

// pathOptions dynamically injects URL parameters
const tenantGroup = new EndpointGroup({
  prefix: "/:tenantId/users",
  pathOptions: async (req) => {
    // req is the NextRequest object (available in proxy mode)
    const tenantId = req?.headers.get("x-tenant-id") || "default"
    return { tenantId }
  },
})

// When tenantId = "acme":
// Method URL: ""       → /acme/users
// Method URL: "/:id"   → /acme/users/:id
```

<Callout variant="info" title="Path resolution order">
Path options from the `Request` level are resolved first, then group-level `pathOptions` are merged on top. Group options take precedence for overlapping keys.
</Callout>

## Multi-Level Path Options

Path options can be defined at both the `Request` and `RequestGroup` level. They are merged together.

```ts
const api = new Datasource({
  baseURL: process.env.API_URL,
  // pathOptions at Request level apply to ALL groups
  pathOptions: async (req) => ({
    version: "v2",
  }),
  groups: {
    users: new EndpointGroup({
      prefix: "/:version/users",
      // pathOptions at group level are merged with Request-level
      pathOptions: async (req) => ({
        tenantId: req?.headers.get("x-tenant-id"),
      }),
    }),
  },
})

// Both :version and :tenantId are resolved:
// "users:getAll" → /v2/users (version from Request pathOptions)
```

## Query Format

When calling methods in a group, use the `"group:method"` format. Root-level methods (no group) use just the method name.

```ts
// Query format with groups: "group:method"
await request("api")("users:getAll")
await request("api")("users:getById", { id: "123" })

// Query format without groups: "method"
await request("api")("health")
```
