### Step 1: Define your API sources

Create a file with your API definition. Each `EndpointGroup` organizes related endpoints under a URL prefix, and each method declares its response and variables types:

```ts title="api/sources.ts"
import { Datasource, EndpointGroup, Endpoint } from "@noverax/nexus/datasource"

// 1. Define response & request types
type User = { id: number; name: string; email: string; role: string }
type UserList = { data: User[]; total: number }

// 2. Create an EndpointGroup with typed methods
class Users extends EndpointGroup {
  constructor() {
    super({ prefix: "users" })
  }

  getAll: Endpoint<UserList, { page?: number; limit?: number }> = ({ get }) => get()
  getById: Endpoint<User, { id: number }> = ({ get }) => get(":id")
  create: Endpoint<User, { name: string; email: string; role: string }> = ({ post }) => post()
  update: Endpoint<User, { id: number; name: string; email: string }> = ({ put }) => put(":id")
  remove: Endpoint<null, { id: number }> = ({ del }) => del(":id")
}

// 3. Create a Datasource with groups and base URL
const groups = { users: new Users() }

class MyApi extends Datasource<typeof groups> {
  constructor() {
    super({
      baseURL: process.env.API_URL ?? "http://localhost:4000",
      groups,
    })
  }
}

// 4. Export sources object
export const sources = { myApi: new MyApi() }
export type Sources = typeof sources
```

<Callout variant="info" title="Path parameters">
The `:id` syntax in `get(":id")` automatically extracts `id` from the variables and inserts it into the URL. The remaining variables are sent as query params (GET) or body (POST/PUT/PATCH).
</Callout>

### Step 2: Set up the transport

**Proxy mode (browser -> API route -> backend):**

Create an API route that forwards browser requests to your backend, then create a client with React Query hooks:

```ts title="app/api/request/route.ts"
// app/api/request/route.ts
import { apiRoute } from "@noverax/nexus/next/apiRoute"
import { sources } from "@/api/sources"

export const { POST } = apiRoute(sources)
```

```ts title="lib/request.ts"
// lib/request.ts
import { createQueryRequest } from "@noverax/nexus/query"
import type { Sources } from "@/api/sources"

export const { useQuery, useMutation, request } = createQueryRequest<Sources>()
```

**Direct mode (server -> backend):**

For Server Components and Server Actions, create a direct request client:

```ts title="lib/direct-request.ts"
// app/api/request/route.ts  — already done in Step 2

// lib/direct-request.ts
import { createDirectRequest } from "@noverax/nexus/direct"
import { sources } from "@/api/sources"

export const directRequest = createDirectRequest(sources)
```

<Callout variant="tip" title="Use both modes together">
Most apps use proxy mode for client components and direct mode for server components. Both share the same source definitions.
</Callout>

### Step 3: Use in components

**Client component (proxy + React Query):**

```tsx title="app/users/page.tsx"
"use client"

import { useQuery, useMutation } from "@/lib/request"

export default function UsersPage() {
  // Auto-fetches on mount, returns typed data
  const { data, isLoading } = useQuery("myApi", "users:getAll", {
    variables: { page: 1, limit: 10 },
  })

  // Mutation for creating users
  const { mutate, isPending } = useMutation("myApi", "users:create", {
    onSuccess: (user) => {
      console.log("Created:", user.name) // ← fully typed
    },
  })

  if (isLoading) return <p>Loading…</p>

  return (
    <div>
      <h1>Users ({data?.total})</h1>
      <ul>
        {data?.data.map((user) => (
          <li key={user.id}>{user.name} — {user.email}</li>
        ))}
      </ul>
      <button
        onClick={() => mutate({ name: "Jane", email: "jane@test.com", role: "user" })}
        disabled={isPending}
      >
        Add User
      </button>
    </div>
  )
}
```

**Server component (direct mode):**

```tsx title="app/dashboard/page.tsx"
// app/dashboard/page.tsx — React Server Component
import { directRequest } from "@/lib/direct-request"

export default async function DashboardPage() {
  const users = await directRequest("myApi")("users:getAll", { page: 1, limit: 5 })

  return (
    <div>
      <h1>Dashboard ({users.total} users)</h1>
      <ul>
        {users.data.map((user) => (
          <li key={user.id}>{user.name}</li>
        ))}
      </ul>
    </div>
  )
}
```

### Step 4: Run your app

Start your backend server and Next.js dev server. The proxy route handles serialization, CORS, and cookie forwarding automatically.

<Callout variant="warning" title="Don't forget your backend">
Nexus doesn't replace your backend — it connects to any REST API. Make sure your backend is running at the URL specified in `API_URL`.
</Callout>

## What's happening under the hood

When you call a query like this:

```typescript
useQuery("myApi", "users:getById", { variables: { id: 5 } })
```

1. Nexus parses `"users:getById"` into group `users` and method `getById`
2. In proxy mode, it POSTs to `/api/request` with the source, group, query, and variables
3. The API route finds the `Users` group, calls `getById` with an `HttpClient` that wraps ky
4. The `:id` path parameter is extracted from variables, and the request goes to `GET /users/5`
5. The typed response flows back through React Query to your component
