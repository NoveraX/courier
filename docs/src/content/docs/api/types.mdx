## Overview

All TypeScript types used across the Courier library, organized by category.

## Client Types

### TypedClient

The main typed client returned by `createCaller` and `createProxyCaller`. Provides dot-notation access to groups, `.raw` for full response, and `.with()` for per-request options.

```ts
type TypedClient<T extends DefinedApi<any>, Raw extends boolean = false> = {
  [K in keyof ApiGroups<T>]: GroupClient<ApiGroups<T>[K], Raw>
} & {
  raw: TypedClient<T, true>
  with(options: WithOptions): TypedClient<T, Raw>
}
```

### GroupClient

A group with typed endpoint methods, `.raw` accessor, and `.with()`.

```ts
type GroupClient<G, Raw extends boolean = false> = GroupMethods<G, Raw> & {
  raw: GroupClient<G, true>
  with(options: WithOptions): GroupClient<G, Raw>
}
```

### WithOptions

Options for `.with()` — all `CallOptions` except `rawData` (controlled via `.raw`).

```ts
type WithOptions = Omit<CallOptions, "rawData">
```

### CallerDefaults

Options for `createCaller`.

```ts
type CallerDefaults = {
  timeout?: number
  headers?: Record<string, string>
  context?: Context
  onError?: (error: HTTPError) => void | Promise<void>
  onSuccess?: (data: unknown) => void | Promise<void>
}
```

### ProxyCallerDefaults

Options for `createProxyCaller` and `createProxyQueryCaller`.

```ts
type ProxyPath = {
  http?: string  // default: "/api/request"
  sse?: string   // default: "/api/sse"
  ws?: string    // no default — throws if WS endpoint invoked without it
}

type ProxyCallerDefaults = {
  proxyPath?: ProxyPath
  source?: object
  timeout?: number
  credentials?: RequestCredentials
  headers?: Record<string, string>
  onError?: (error: HTTPError) => void | Promise<void>
  onSuccess?: (data: unknown) => void | Promise<void>
}
```

## Endpoint Definition Types

### Endpoint Types

An endpoint is an inert descriptor, not a callable. One of three types names it,
according to the head it was built from:

```ts
type HttpEndpoint<Response, Variables, Kind extends "query" | "mutation">
type StreamEndpoint<Response, Variables>
type ChannelEndpoint<Response, Variables, Send>

// Any of the three, when the kind does not matter
type AnyEndpoint
```

### Inference Helpers

Read the types back off an endpoint. This is how a caller knows what a method
takes and returns:

```ts
type InferResponse<E extends AnyEndpoint>   // what the endpoint resolves to
type InferVariables<E extends AnyEndpoint>  // what the caller must pass
type InferSend<E extends AnyEndpoint>       // what a channel accepts from the client
type InferKind<E extends AnyEndpoint>       // "query" | "mutation" | "stream" | "channel"
```

### Wire Surfaces

The object `.via()` hands the closure. Name these to lift a closure out of the
chain into a function of its own:

```ts
interface HttpWire {
  get(path?: string, variables?: unknown): WireCall
  post(path?: string, body?: unknown): WireCall
  put(path?: string, body?: unknown): WireCall
  patch(path?: string, body?: unknown): WireCall
  del(path?: string, variables?: unknown): WireCall
  head(path?: string, variables?: unknown): WireCall
}

interface StreamWire {
  get(path?: string, variables?: unknown): WireCall
}

interface ChannelWire {
  connect(path?: string, variables?: unknown): WireCall
}

type HttpProbe<Variables> = (http: HttpWire, variables: Variables) => WireCall
type StreamProbe<Variables> = (sse: StreamWire, variables: Variables) => WireCall
type ChannelProbe<Variables> = (ws: ChannelWire, variables: Variables) => WireCall
```

```ts
import type { HttpProbe } from "@noverax/courier/define"

const byId: HttpProbe<{ id: string }> = (http, vars) => http.get(":id", vars)

const getById = query.variables<{ id: string }>().response<User>().via(byId)
```

## Response Types

### HttpResponse

```ts
type HttpResponse<T = unknown> = {
```

## Options Types

### CallOptions

Per-request options passed when calling an endpoint.

```ts
type CallOptions<R extends boolean = false> = SimpleOptions & {
  rawData?: R
  forwardVariables?: boolean
  stream?: boolean
  method?: "GET" | "POST"
  onUploadProgress?: (progress: UploadProgress) => void
}
```

### UploadProgress

```ts
type UploadProgress = {
  percent: number           // 0 to 1
  transferredBytes: number
  totalBytes: number
}
```

### MethodParameters

```ts
type MethodParameters<T extends HttpMethod> = T extends "head"
  ? [url?: string, options?: MethodOptions<T>]
  : [url?: string, variables?: Record<string, unknown>, options?: MethodOptions<T>]
```

```ts

  responseParser?: (response: Response) => Promise<unknown>

  defaultQueryParams?: Record<string, unknown>
  allowedHeaders?: string[]
```

```ts
type GroupOptions = MergeOptions & {
```

## Query Utility Types

Internal utility types that power the type inference engine:

```ts
// Extracts all valid endpoint keys for a source
type EndpointKey<T> = FunctionKeys<T> | GroupedEndpoints<Groups<T>>
// Result: "methodName" | "group:methodName"

// Resolves the endpoint an endpoint key names
type ResolveEndpoint<T, Q> = /* resolves to the endpoint descriptor */

// Extracts response type from an endpoint
type ExtractResponse<E> = E extends AnyEndpoint ? InferResponse<E> : unknown

// Extracts variables type from an endpoint
type ExtractVariables<E> = E extends AnyEndpoint ? InferVariables<E> : unknown
```

<Callout variant="info" title="Type inference">
These utility types are what make Courier fully type-safe. When you call `caller.users.getById({ id })`, TypeScript resolves the group, endpoint, variables, and response types automatically.
</Callout>

## React Query Types

```ts
type QueryOptions<R, V> = WithVariables<V> &
  Omit<UseQueryOptions<R, HTTPError, R, QueryKey>, "queryKey" | "queryFn"> & {
    requestOptions?: CallOptions
  }

type MutationOptions<R, V> = Omit<
  UseMutationOptions<R, HTTPError, MutationVariables<V>>,
  "mutationFn"
> & {
  requestOptions?: CallOptions
}

type MutationVariables<V> = [V] extends [undefined]
  ? void
  : {} extends V ? void | V : V

type WithVariables<V> = [V] extends [undefined]
  ? { variables?: undefined }
  : {} extends V
    ? { variables?: V }
    : { variables: V }
```
