export const metadata = { title: "HTTPError", description: "Custom error class for HTTP responses with typed data, status code, and headers. Includes the isHTTPError type guard." }

## Import

```ts
import { HTTPError, isHTTPError } from "@noverax/nexus"
```

## HTTPError Class

```ts
class HTTPError<R = unknown> extends Error {
  status: number
  headers: Headers
  data: R

  constructor(response: HttpResponse<R>)
}
```

## Properties

```ts
// HTTPError properties:
error.name       // "HTTPError"
error.message    // "HTTP {status}" (e.g., "HTTP 404")
error.status     // HTTP status code (e.g., 404)
error.headers    // Response headers (Headers object)
error.data       // Parsed response body (typed as R)
```

- `status` — the HTTP status code. `0` for network errors (DNS failure, connection refused, etc.).
- `headers` — the response `Headers` object. Empty `Headers` for network errors.
- `data` — the parsed response body. Typed as `R` (generic, defaults to `unknown`).
- `name` — always `"HTTPError"`.
- `message` — `"HTTP {status}"` (e.g., `"HTTP 404"`).

## isHTTPError

Type guard function to check if an error is an HTTPError.

```ts
function isHTTPError<R = unknown>(error: unknown): error is HTTPError<R>
```

<Callout variant="tip" title="Prefer isHTTPError over instanceof">
Use `isHTTPError(error)` instead of `error instanceof HTTPError`. The type guard works reliably across module boundaries and different package versions.
</Callout>

```ts
// isHTTPError vs instanceof
// Both work, but isHTTPError is the recommended approach:

// ✅ Recommended — works across module boundaries
if (isHTTPError(error)) { ... }

// ⚠️ Works, but may fail across different module instances
if (error instanceof HTTPError) { ... }
```

## Basic Usage

```ts
import { isHTTPError } from "@noverax/nexus"

try {
  const user = await request("api")("users:getById", { id: "999" })
} catch (error) {
  if (isHTTPError(error)) {
    console.error(`HTTP ${error.status}:`, error.data)
    // error.status  → 404
    // error.headers → Headers { content-type: "application/json" }
    // error.data    → { message: "User not found" }
  }
}
```

## Typed Error Data

Pass a type parameter to `isHTTPError` to type the error's `data` property.

```ts
// Type the error data for better inference
type ErrorResponse = {
  success: false
  message: string
  code?: string
}

try {
  await request("api")("users:getById", { id: "999" })
} catch (error) {
  if (isHTTPError<ErrorResponse>(error)) {
    // error.data is typed as ErrorResponse
    console.error(error.data.message)  // "User not found"
    console.error(error.data.code)     // "USER_NOT_FOUND"
  }
}
```

## Status Code Handling

```ts
try {
  await request("api")("users:getAll")
} catch (error) {
  if (isHTTPError(error)) {
    switch (error.status) {
      case 400:
        console.error("Bad request:", error.data)
        break
      case 401:
        redirectToLogin()
        break
      case 403:
        showForbiddenMessage()
        break
      case 404:
        showNotFound()
        break
      case 429:
        console.warn("Rate limited, retry later")
        break
      default:
        if (error.status >= 500) {
          console.error("Server error:", error.data)
        }
    }
  }
}
```

## Global Error Handler

Use `onError` in `createClientRequest` or `createDirectRequest` to handle errors globally.

```ts
// Global error handler via createClientRequest
const request = createClientRequest<Sources>({
  onError: (error) => {
    // Called for every HTTPError across all requests
    if (error.status === 401) {
      // Token expired — redirect to login
      window.location.href = "/login"
    }

    if (error.status >= 500) {
      // Log server errors
      reportError(error)
    }
  },
})
```

## Source-Level Error Hook

The `onError` hook on `Request` receives both the request metadata and the error.

```ts
// Source-level error hook via Request options
const api = new Datasource({
  baseURL: process.env.API_URL,
  onError: (info, error) => {
    console.error(
      `[${info.source}:${info.group}:${info.method}] ` +
      `HTTP ${error.status}: ${JSON.stringify(error.data)}`
    )
    return undefined
  },
})
```

## Network Errors

Network-level errors (DNS failures, connection refused, timeouts at the TCP level) are converted to `HTTPError` with `status: 0`.

```ts
// Network errors (DNS failure, connection refused, etc.)
// are converted to HTTPError with status: 0
try {
  await request("api")("users:getAll")
} catch (error) {
  if (isHTTPError(error)) {
    if (error.status === 0) {
      // Network error — no HTTP response received
      console.error("Network error:", error.data)
      // error.data → { success: false, message: "...", code: "ECONNREFUSED" }
    }
  }
}
```

## Raw Data Mode

When using `rawData: true`, the error data is extracted from the wrapper object if present.

```ts
// With rawData mode, error.data contains the raw response data
// (without the wrapper object)
const result = await request("api", { rawData: true })(
  "users:getById",
  { id: "999" },
)
// If the response is { data: { message: "Not found" } },
// rawData mode extracts error.data as { message: "Not found" }
```
