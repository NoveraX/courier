## Import

```ts
import { HTTPError, isHTTPError } from "@noverax/nexus/error"
```

## HTTPError Class

```ts
class HTTPError<R = unknown> extends Error {
  status: number
  headers: Headers
  data: R

  constructor(response: HttpResponse<R>)
}
```

## Properties

```ts
error.name       // "HTTPError"
error.message    // "HTTP {status}" (e.g., "HTTP 404")
error.status     // HTTP status code (e.g., 404)
error.headers    // Response headers (Headers object)
error.data       // Parsed response body (typed as R)
```

- `status` — the HTTP status code. `0` for network errors (DNS failure, connection refused).
- `headers` — the response `Headers` object. Empty `Headers` for network errors.
- `data` — the parsed response body. Typed as `R` (defaults to `unknown`).
- `name` — always `"HTTPError"`.
- `message` — `"HTTP {status}"` (e.g., `"HTTP 404"`).

## isHTTPError

Type guard function to check if an error is an HTTPError:

```ts
function isHTTPError<R = unknown>(error: unknown): error is HTTPError<R>
```

<Callout variant="tip" title="Prefer isHTTPError over instanceof">
Use `isHTTPError(error)` instead of `error instanceof HTTPError`. The type guard works reliably across module boundaries and different package versions.
</Callout>

## Basic Usage

```ts
import { isHTTPError } from "@noverax/nexus/error"

try {
  const user = await call("api")("users:getById", { id: "999" })
} catch (error) {
  if (isHTTPError(error)) {
    console.error(`HTTP ${error.status}:`, error.data)
  }
}
```

## Typed Error Data

Pass a type parameter to `isHTTPError` to type the error's `data`:

```ts
type ErrorResponse = {
  success: false
  message: string
  code?: string
}

try {
  await call("api")("users:getById", { id: "999" })
} catch (error) {
  if (isHTTPError<ErrorResponse>(error)) {
    console.error(error.data.message)  // "User not found"
    console.error(error.data.code)     // "USER_NOT_FOUND"
  }
}
```

## Status Code Handling

```ts
try {
  await call("api")("users:getAll")
} catch (error) {
  if (isHTTPError(error)) {
    switch (error.status) {
      case 400: console.error("Bad request:", error.data); break
      case 401: redirectToLogin(); break
      case 403: showForbiddenMessage(); break
      case 404: showNotFound(); break
      case 429: console.warn("Rate limited, retry later"); break
      default:
        if (error.status >= 500) console.error("Server error:", error.data)
    }
  }
}
```

## Middleware-Based Error Handling

Use middleware to handle errors globally across all requests:

```ts
import { defineMiddleware } from "@noverax/nexus/middleware"
import { isHTTPError } from "@noverax/nexus/error"

const errorMiddleware = defineMiddleware(async (ctx, next) => {
  try {
    return await next()
  } catch (error) {
    if (isHTTPError(error) && error.status === 401) {
      // redirect to login, refresh token, etc.
    }
    throw error
  }
})
```

## Network Errors

Network-level errors (DNS failures, connection refused, timeouts) are converted to `HTTPError` with `status: 0`:

```ts
try {
  await call("api")("users:getAll")
} catch (error) {
  if (isHTTPError(error) && error.status === 0) {
    console.error("Network error:", error.data)
  }
}
```
