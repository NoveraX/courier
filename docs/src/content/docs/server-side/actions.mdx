## Overview

`createAction` from `@noverax/nexus-next` creates type-safe server actions that integrate with React 19's `useActionState`. It handles FormData parsing, header forwarding, revalidation, and error handling automatically.

```ts title="app/actions/users.ts"
"use server"

import { createAction } from "@noverax/nexus-next"
import { sources } from "@/api"

export const createUser = createAction(sources)("api")("users:create", {
  revalidatePath: "/users",
})
```

## API signature

`createAction` is a triple-curried function: sources, then source name, then endpoint key with options:

```ts
import { createAction } from "@noverax/nexus-next"

function createAction(sources: Sources)
  (source: string)
  (endpoint: string, options?: ActionOptions): ServerAction
```

## ActionOptions

Configure server action behavior with these options:

```ts
type ActionOptions = {
  // Path(s) to revalidate after successful mutation
  revalidatePath?: string | string[]

  // Tag(s) to revalidate after successful mutation
  revalidateTag?: string | string[]

  // Cache profile for revalidateTag (Next.js 16+)
  cacheProfile?: string | { expire?: number }

  // Custom FormData parser
  // Default: Object.fromEntries
  // Use "raw" to pass FormData directly (file uploads)
  parse?: "raw" | ((formData: FormData) => Record<string, unknown>)

  // Forward incoming request headers to the API (default: true)
  headers?: boolean
}
```

## ActionState

Every server action returns a consistent `ActionState` object:

```ts
type ActionState<T = unknown> = {
  data: T | null
  error: {
    message: string
    status?: number
    issues?: readonly ValidationIssue[]  // Standard Schema validation errors
  } | null
}
```

## Basic usage

Create a server action file with `"use server"` and define your actions:

```ts title="app/actions/users.ts"
"use server"

import { createAction } from "@noverax/nexus-next"
import { sources } from "@/api"

const action = createAction(sources)("api")

// Simple mutation with path revalidation
export const createUser = action("users:create", {
  revalidatePath: "/users",
})

// Mutation with tag revalidation
export const updateUser = action("users:update", {
  revalidateTag: "users",
})

// Mutation with multiple revalidation paths
export const deleteUser = action("users:delete", {
  revalidatePath: ["/users", "/dashboard"],
})
```

## Using with useActionState

Server actions created by `createAction` are designed for React 19's `useActionState`:

```ts title="app/users/create-form.tsx"
"use client"

import { useActionState } from "react"
import { createUser } from "@/app/actions/users"

export function CreateUserForm() {
  const [state, formAction, isPending] = useActionState(createUser, {
    data: null,
    error: null,
  })

  return (
    <form action={formAction}>
      <input name="name" placeholder="Name" required />
      <input name="email" type="email" placeholder="Email" required />
      <button type="submit" disabled={isPending}>
        {isPending ? "Creating..." : "Create User"}
      </button>

      {state.error && <p className="text-red-500">{state.error.message}</p>}

      {state.error?.issues && (
        <ul>
          {state.error.issues.map((issue, i) => (
            <li key={i}>{issue.message}</li>
          ))}
        </ul>
      )}

      {state.data && <p className="text-green-500">Created: {state.data.name}</p>}
    </form>
  )
}
```

<Callout variant="info" title="Automatic headers">
By default, `createAction` forwards the incoming request headers (including cookies) to your backend. Set `headers: false` to disable this behavior.
</Callout>

## File uploads

For file upload actions, use `parse: "raw"` to pass FormData directly to your endpoint without converting to a plain object:

```ts title="app/actions/files.ts"
"use server"

import { createAction } from "@noverax/nexus-next"
import { sources } from "@/api"

export const uploadAvatar = createAction(sources)("api")("users:uploadAvatar", {
  parse: "raw",
  revalidatePath: "/profile",
})
```

```ts title="app/profile/avatar-form.tsx"
"use client"

import { useActionState } from "react"
import { uploadAvatar } from "@/app/actions/files"

export function AvatarForm() {
  const [state, formAction, isPending] = useActionState(uploadAvatar, {
    data: null,
    error: null,
  })

  return (
    <form action={formAction}>
      <input name="avatar" type="file" accept="image/*" required />
      <button type="submit" disabled={isPending}>
        {isPending ? "Uploading..." : "Upload"}
      </button>
    </form>
  )
}
```

## Custom FormData parsing

Provide a custom parser when `Object.fromEntries` is not sufficient (e.g., checkboxes, multi-select, or nested data):

```ts title="app/actions/settings.ts"
"use server"

import { createAction } from "@noverax/nexus-next"
import { sources } from "@/api"

export const updateSettings = createAction(sources)("api")("settings:update", {
  parse: (formData) => {
    return {
      theme: formData.get("theme"),
      notifications: formData.get("notifications") === "on",
      tags: formData.getAll("tags"),
    }
  },
  revalidateTag: "settings",
})
```

## Error handling

`createAction` automatically catches errors and returns them in the `ActionState.error` field:

```typescript
// Error handling is automatic:

// HTTP errors from your backend:
// { data: null, error: { message: "User not found", status: 404 } }

// Validation errors (Standard Schema):
// { data: null, error: { message: "Validation failed", status: 422, issues: [...] } }

// Unknown errors:
// { data: null, error: { message: "Unknown error" } }
```

<Callout variant="tip" title="No try/catch needed">
Unlike manual server actions, `createAction` handles errors internally and always returns a result object. Your client components can check `state.error` without wrapping calls in try/catch.
</Callout>

## Manual server actions

If you need more control than `createAction` provides, use `createCaller` directly in a server action:

```ts title="app/actions/users.ts"
"use server"

import { headers } from "next/headers"
import { revalidatePath } from "next/cache"
import { caller } from "@/lib/caller"

type CreateUserInput = { name: string; email: string; role: string }

export async function createUser(data: CreateUserInput) {
  const user = await caller("api")(
    "users:create",
    data,
    { headers: await headers() }
  )

  revalidatePath("/dashboard")
  return user
}
```

<Callout variant="warning" title="Always forward headers">
When using `createCaller` in server actions, always pass `{ headers: await headers() }` as the context argument so authentication cookies reach your backend. `createAction` does this automatically.
</Callout>

## Server Actions vs useMutation

Both Server Actions and React Query's `useMutation` can handle mutations. They serve different use cases:

- **Server Actions** — best for form submissions, progressive enhancement, and when you want Next.js to handle revalidation
- **useMutation** — best for optimistic updates, cache invalidation via query keys, and when you need React Query's retry/rollback features

```ts title="app/users/create-button.tsx"
"use client"

import { useMutation } from "@/lib/caller"

export function CreateUserButton() {
  // useMutation calls the proxy endpoint, not a Server Action
  // Use this when you want React Query's cache management
  const { mutate, isPending } = useMutation("api", "users:create", {
    onSuccess: (user) => {
      console.log("Created:", user.name)
    },
  })

  return (
    <button
      onClick={() => mutate({ name: "Jane", email: "jane@test.com", role: "user" })}
      disabled={isPending}
    >
      {isPending ? "Creating..." : "Add User"}
    </button>
  )
}
```
