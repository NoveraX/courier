export const metadata = { title: "Request", description: "Class that defines a data source with base URL, options, lifecycle hooks, and optional request groups." }

## Import

```ts
import { Datasource } from "@noverax/nexus/datasource"
```

## Signature

```ts
class Datasource<T extends RequestGroupType | undefined = undefined> {
  constructor(public options?: RequestOptions<T>)
}
```

## Constructor

### RequestOptions

`RequestOptions` extends `BaseOptions` and adds an optional `groups` property when request groups are used.

```ts
type RequestOptions<T> = BaseOptions &
  (T extends RequestGroupType ? { groups: T } : {})

type RequestGroupType = Record<string, EndpointGroup>
```

### BaseOptions

The core configuration for a data source. Includes Nexus-specific options and all `ky` options (timeout, retry, headers, hooks, etc.).

```ts
type BaseOptions = MergeOptions & KyOptions & {
  // Base URL for all requests from this source
  baseURL?: string

  // Response parsing (default: "json")
  responseType?: "json" | "text" | "blob" | "arrayBuffer" | "formData"

  // Custom response parser (overrides responseType)
  responseParser?: (response: Response) => Promise<unknown>

  // Default body data merged into every POST/PUT/PATCH request
  defaultBody?: Record<string, unknown>

  // Default query parameters merged into every request
  defaultQueryParams?: Record<string, unknown>

  // Additional header names to preserve through the proxy
  allowedHeaders?: string[]

  // Custom body serializer (e.g., for URL-encoded forms)
  bodySerializer?: (data: Record<string, unknown>) => BodyInit

  // Custom search params serializer (e.g., qs)
  searchParamsSerializer?: { stringify: (obj) => string; parse: (str) => Record<string, unknown> }

  // Enable AbortController for request deduplication
  deduplicate?: boolean

  // Lifecycle hooks
  onRequest?: (info: HookInfo) => HookInfo | undefined | Promise<HookInfo | undefined>
  onResponse?: (
    info: HookInfo,
    response: { status: number; data: unknown },
  ) => { status: number; data: unknown } | undefined | Promise<...>
  onError?: (info: HookInfo, error: HTTPError) => undefined | Promise<undefined>
}
```

## Basic Usage

### Without Groups

```ts title="api/sources.ts"
import { Datasource } from "@noverax/nexus/datasource"

// Simple source without groups
const api = new Datasource({
  baseURL: "https://api.example.com/v1",
  timeout: 10000,
  headers: {
    Authorization: `Bearer ${process.env.API_SECRET}`,
  },
})
```

### With Groups

```ts title="api/sources.ts"
import { Datasource, EndpointGroup } from "@noverax/nexus/datasource"

// Source with request groups
const api = new Datasource({
  baseURL: process.env.API_URL,
  timeout: 10000,
  responseType: "json",
  groups: {
    users: new EndpointGroup({ prefix: "/users" }),
    posts: new EndpointGroup({ prefix: "/posts" }),
    files: new EndpointGroup({ prefix: "/files" }),
  },
})
```

## Options Reference

### baseURL

Sets the base URL (prefix) for all requests from this source. Maps to `ky`'s `prefixUrl` option.

### responseType

Controls response parsing. Supported values: `"json"` (default), `"text"`, `"blob"`, `"arrayBuffer"`, `"formData"`. A `204` status always returns `null`.

### responseParser

Custom response parser that overrides `responseType`. Receives the raw `Response` object. Useful for XML, Protocol Buffers, or any format not covered by `responseType`:

```ts
const api = new Datasource({
  baseURL: process.env.API_URL,
  responseParser: async (response) => {
    const text = await response.text()
    return new DOMParser().parseFromString(text, "text/xml")
  },
})
```

### defaultBody

Object merged into every POST/PUT/PATCH request body. Useful for API keys or common parameters.

```ts
// defaultBody is merged into every POST/PUT/PATCH request body
const api = new Datasource({
  baseURL: process.env.API_URL,
  defaultBody: {
    apiKey: process.env.API_KEY,
    version: "v2",
  },
})

// POST /users with body: { apiKey: "...", version: "v2", name: "Alice" }
// The apiKey and version are automatically included
```

### defaultQueryParams

Object merged into every request's search parameters.

```ts
// defaultQueryParams is merged into every request's search parameters
const api = new Datasource({
  baseURL: process.env.API_URL,
  defaultQueryParams: {
    locale: "en",
    format: "json",
  },
})

// GET /users → GET /users?locale=en&format=json
// GET /users?page=1 → GET /users?locale=en&format=json&page=1
```

### allowedHeaders

An array of header names that should be preserved when forwarding requests through the proxy. By default, only standard headers are forwarded.

## Lifecycle Hooks

Three hooks give you full control over the request lifecycle at the source level.

```ts title="api/sources.ts"
const api = new Datasource({
  baseURL: process.env.API_URL,

  // Called before every request — can modify or cancel
  onRequest: (info) => {
    console.log(`[${info.source}] ${info.group}:${info.method}`)
    return info // return modified info or undefined to continue
  },

  // Called after every successful response
  onResponse: (info, response) => {
    console.log(`[${info.source}] ${response.status}`)
    return response // return modified response or undefined to use original
  },

  // Called on every error
  onError: (info, error) => {
    console.error(`[${info.source}] Error ${error.status}:`, error.data)
    return undefined
  },
})
```

```ts
type HookInfo = {
  source: string      // source name (e.g., "api")
  group?: string       // group name (e.g., "users")
  method: string       // method name (e.g., "getById")
  variables?: unknown  // request variables
}
```

<Callout variant="tip" title="Hook vs ky hooks">
`onRequest`, `onResponse`, and `onError` are Nexus-level hooks with access to source/group/query metadata. For low-level HTTP hooks, use the `hooks` option from `ky`.
</Callout>

## Custom Encoders

Override the default JSON body encoding and search params serialization.

```ts
// Custom body serializer — e.g., URL-encoded form data
const api = new Datasource({
  baseURL: process.env.API_URL,
  bodySerializer: (data) => {
    return new URLSearchParams(
      Object.entries(data).map(([k, v]) => [k, String(v)])
    ).toString()
  },
})

// Custom search params serializer — e.g., using qs
import qs from "qs"
const api2 = new Datasource({
  baseURL: process.env.API_URL,
  searchParamsSerializer: qs,
})
```

## Request Abort

Enable `deduplicate` for automatic request deduplication. When a duplicate request is fired while one is in-flight, the previous request is aborted.

```ts
// Enable request deduplication via AbortController
const api = new Datasource({
  baseURL: process.env.API_URL,
  deduplicate: true,
})

// If the same request is fired while one is in-flight,
// the previous request is automatically aborted
```

## ky Options

All `ky` options (timeout, retry, headers, hooks, etc.) are supported and passed through to `ky.create()`.

```ts
// All ky options are supported — they pass through to ky.create()
const api = new Datasource({
  baseURL: process.env.API_URL,

  // Timeout per request (ms)
  timeout: 10000,

  // Retry on failure
  retry: { limit: 3, methods: ["get"] },

  // Custom headers
  headers: {
    "Content-Type": "application/json",
    Authorization: `Bearer ${process.env.API_TOKEN}`,
  },

  // ky hooks (low-level — prefer onRequest/onResponse/onError above)
  hooks: {
    beforeRequest: [(request) => { /* ... */ }],
    afterResponse: [(request, options, response) => { /* ... */ }],
  },
})
```
